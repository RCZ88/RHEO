```ts
const STEPS = 32, STEP_MS = 20;                    // 640ms
for (let i = 0; i <= STEPS; i++) {
  const t = i / STEPS;
  const th = (180 + 360 * t) * Math.PI / 180;      // increasing θ = CW in y-down screen space
  const r  = R0k + (R1k - R0k) * Math.min(1, t * STEPS / 3);  // radial ramp over first 3 steps
```
